  - id: "V1042.002"
    name: "Proof requirement"
    description: "Drop findings that cite no line, input or behaviour."
  - id: "V1042.003"
    name: "Run to confirm"
    description: "Write and run a test or script that demonstrates the failure."
  - id: "V1042.004"
    name: "Atomic findings"
    description: "Split findings that make more than one claim so each is verified separately."
